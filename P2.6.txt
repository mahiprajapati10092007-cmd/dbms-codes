P2.6  Zomato Delivery Price Engine   [Medium]  [IF-CASE combo, Boolean flags, multiple surcharges] Problem Statement:  Build a Zomato-style delivery charge calculator. Base charge: <3km=Free, 38km=Rs.29, 8-15km=Rs.49, >15km=Rs.79. Surcharges applied using CASE: Rain=+20%, Late Night (10pm-6am)=+15%, Festival Day=+10%, Peak Hour (12-2pm or 7-9pm)=+10%. Discounts using IF: order_value>499=delivery Free (overrides everything), First Order=50% off delivery. Print full receipt: base, each surcharge/discount, final delivery, food total, grand total. Round all to nearest rupee. 


//answer

SET SERVEROUTPUT ON;

DECLARE
    v_food_value NUMBER := 450;
    v_distance NUMBER := 5;
    v_rain CHAR(1) := 'Y';
    v_late_night CHAR(1) := 'N';
    v_festival CHAR(1) := 'N';
    v_peak_hour CHAR(1) := 'Y';
    v_first_order CHAR(1) := 'N';

    v_base NUMBER := 0;
    v_rain_charge NUMBER := 0;
    v_late_charge NUMBER := 0;
    v_festival_charge NUMBER := 0;
    v_peak_charge NUMBER := 0;
    v_final_delivery NUMBER;
    v_grand_total NUMBER;
BEGIN
    IF v_distance < 3 THEN
        v_base := 0;
    ELSIF v_distance <= 8 THEN
        v_base := 29;
    ELSIF v_distance <= 15 THEN
        v_base := 49;
    ELSE
        v_base := 79;
    END IF;

    CASE v_rain
        WHEN 'Y' THEN v_rain_charge := v_base * 0.20;
        ELSE v_rain_charge := 0;
    END CASE;

    CASE v_late_night
        WHEN 'Y' THEN v_late_charge := v_base * 0.15;
        ELSE v_late_charge := 0;
    END CASE;

    CASE v_festival
        WHEN 'Y' THEN v_festival_charge := v_base * 0.10;
        ELSE v_festival_charge := 0;
    END CASE;

    CASE v_peak_hour
        WHEN 'Y' THEN v_peak_charge := v_base * 0.10;
        ELSE v_peak_charge := 0;
    END CASE;

    v_final_delivery := v_base + v_rain_charge + v_late_charge
                        + v_festival_charge + v_peak_charge;

    IF v_food_value > 499 THEN
        v_final_delivery := 0;
    ELSIF v_first_order = 'Y' THEN
        v_final_delivery := v_final_delivery * 0.50;
    END IF;

    v_grand_total := v_food_value + v_final_delivery;

    DBMS_OUTPUT.PUT_LINE('Food Total: Rs.' || v_food_value);
    DBMS_OUTPUT.PUT_LINE('Base Delivery: Rs.' || v_base);
    DBMS_OUTPUT.PUT_LINE('Rain Charge: Rs.' || ROUND(v_rain_charge));
    DBMS_OUTPUT.PUT_LINE('Late Night Charge: Rs.' || ROUND(v_late_charge));
    DBMS_OUTPUT.PUT_LINE('Festival Charge: Rs.' || ROUND(v_festival_charge));
    DBMS_OUTPUT.PUT_LINE('Peak Hour Charge: Rs.' || ROUND(v_peak_charge));
    DBMS_OUTPUT.PUT_LINE('Final Delivery: Rs.' || ROUND(v_final_delivery));
    DBMS_OUTPUT.PUT_LINE('Grand Total: Rs.' || ROUND(v_grand_total));
END;
