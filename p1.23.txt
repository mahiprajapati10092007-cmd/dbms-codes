the value, (3) CONSTANT declared without a value and assigned later, (4) missing semicolon after 
a declaration, (5) SELECT without INTO inside PL/SQL. After fixing, the block must compile
 and print your bonus.


DECLARE
    v_roll NUMBER := 34;

    v_qty NUMBER;
    v_price NUMBER;
    v_coupon NUMBER := NULL;

    v_gross NUMBER;
    v_disc NUMBER;
    v_taxable NUMBER;
    v_cgst NUMBER;
    v_sgst NUMBER;
    v_total NUMBER;
BEGIN
    v_qty := MOD(v_roll, 5) + 2;
    v_price := v_roll * 100 + 400;

    v_gross := v_qty * v_price;

    v_disc :=
        v_gross *
        NVL(v_coupon, 0) / 100;

    v_taxable :=
        v_gross - v_disc;

    v_cgst :=
        v_taxable * 9 / 100;

    v_sgst :=
        v_taxable * 9 / 100;

    v_total := ROUND(
        v_taxable +
        v_cgst +
        v_sgst
    );

    DBMS_OUTPUT.PUT_LINE(
        '----- KIRANA STORE BILL -----'
    );
    DBMS_OUTPUT.PUT_LINE(
        'Quantity: ' || v_qty
    );
    DBMS_OUTPUT.PUT_LINE(
        'Price: Rs.' || v_price
    );
    DBMS_OUTPUT.PUT_LINE(
        'Gross: Rs.' || v_gross
    );
    DBMS_OUTPUT.PUT_LINE(
        NVL2(
            v_coupon,
            'Coupon applied: ' ||
            v_coupon || '%',
            'No coupon'
        )
    );
    DBMS_OUTPUT.PUT_LINE(
        'Discount: Rs.' || v_disc
    );
    DBMS_OUTPUT.PUT_LINE(
        'CGST: Rs.' || v_cgst
    );
    DBMS_OUTPUT.PUT_LINE(
        'SGST: Rs.' || v_sgst
    );
    DBMS_OUTPUT.PUT_LINE(
        'Final Total: Rs.' || v_total
    );
END;
