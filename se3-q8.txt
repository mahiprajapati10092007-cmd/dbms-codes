SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_pub(p_country VARCHAR2) IS
        SELECT pub_name, city
        FROM publisher
        WHERE UPPER(country) = UPPER(p_country);

    v_country VARCHAR2(30) := '&country';
    v_found BOOLEAN := FALSE;
BEGIN
    FOR r IN c_pub(v_country) LOOP
        v_found := TRUE;

        DBMS_OUTPUT.PUT_LINE(
            r.pub_name || ' - ' || r.city
        );
    END LOOP;

    -- %ROWCOUNT cannot be checked after a cursor FOR loop
    -- because the FOR loop automatically closes the cursor.

    IF NOT v_found THEN
        DBMS_OUTPUT.PUT_LINE(
            'No publisher found in ' || v_country
        );
    END IF;
END;
/
