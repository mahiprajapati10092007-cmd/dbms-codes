P2.5  Fibonacci, Primes and GCD — WHILE Loop   [Medium]  [WHILE LOOP, MOD, mathematical algorithms] Problem Statement:  Part A: Print first N Fibonacci numbers (N = 5 + roll_no MOD 8). Part B: Check if the Nth Fibonacci number is prime using inner WHILE loop. Part C: Print all primes between 1 and 100. Count them. Part D: Find GCD of two numbers using Euclidean algorithm (WHILE b!=0 LOOP r:=MOD(a,b); a:=b; b:=r; END LOOP). Part E: Check if a number is a perfect number (sum of divisors = number, e.g. 6=1+2+3). 

//answer

SET SERVEROUTPUT ON;

DECLARE
    v_roll_no NUMBER := 34;
    n NUMBER := 5 + MOD(v_roll_no, 8);
    a NUMBER := 0;
    b NUMBER := 1;
    c NUMBER;
    v_nth NUMBER := 0;
    i NUMBER := 1;
    j NUMBER;
    v_prime BOOLEAN := TRUE;
    v_count NUMBER := 0;

    x NUMBER := 48;
    y NUMBER := 18;
    r NUMBER;

    v_number NUMBER := 28;
    v_sum NUMBER := 0;
BEGIN
    -- Part A: Fibonacci
    DBMS_OUTPUT.PUT_LINE('First ' || n || ' Fibonacci Numbers:');

    WHILE i <= n LOOP
        DBMS_OUTPUT.PUT_LINE(a);
        v_nth := a;
        c := a + b;
        a := b;
        b := c;
        i := i + 1;
    END LOOP;

    -- Part B: Prime check
    IF v_nth < 2 THEN
        v_prime := FALSE;
    ELSE
        j := 2;

        WHILE j <= SQRT(v_nth) LOOP
            IF MOD(v_nth, j) = 0 THEN
                v_prime := FALSE;
            END IF;
            j := j + 1;
        END LOOP;
    END IF;

    IF v_prime THEN
        DBMS_OUTPUT.PUT_LINE(v_nth || ' is Prime');
    ELSE
        DBMS_OUTPUT.PUT_LINE(v_nth || ' is Not Prime');
    END IF;

    -- Part C: Primes 1 to 100
    DBMS_OUTPUT.PUT_LINE('Prime Numbers from 1 to 100:');

    FOR i IN 2..100 LOOP
        v_prime := TRUE;

        FOR j IN 2..FLOOR(SQRT(i)) LOOP
            IF MOD(i, j) = 0 THEN
                v_prime := FALSE;
                EXIT;
            END IF;
        END LOOP;

        IF v_prime THEN
            DBMS_OUTPUT.PUT_LINE(i);
            v_count := v_count + 1;
        END IF;
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Total Primes: ' || v_count);

    -- Part D: GCD
    WHILE y <> 0 LOOP
        r := MOD(x, y);
        x := y;
        y := r;
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('GCD: ' || x);

    -- Part E: Perfect Number
    FOR i IN 1..v_number - 1 LOOP
        IF MOD(v_number, i) = 0 THEN
            v_sum := v_sum + i;
        END IF;
    END LOOP;

    IF v_sum = v_number THEN
        DBMS_OUTPUT.PUT_LINE(v_number || ' is a Perfect Number');
    ELSE
        DBMS_OUTPUT.PUT_LINE(v_number || ' is not a Perfect Number');
    END IF;
END;
