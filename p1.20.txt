P1.20  Railway PNR and Ticket Generator   [Hard]  [String functions, LPAD, TO_CHAR, date 
arithmetic, MONTHS_BETWEEN] Problem Statement:  Build an IRCTC-style ticket generator. 
Inputs: passenger name, dob, roll_no. Steps: (1) PNR = 'GJ' || TO_CHAR(SYSDATE,'YYMM') || 
LPAD(roll_no, 4, '0'), (2) journey date = SYSDATE + (roll_no MOD 30 + 1), printed as 
Day, DD-MON-YYYY, (3) base fare = roll_no * 50 + 200, (4) age = 
TRUNC(MONTHS_BETWEEN(SYSDATE, dob)/12); senior citizens (60+) get 40% concession — compute 
final fare with NVL2-style logic, (5) print the full ticket aligned with RPAD/LPAD like a 
printed counter ticket.


DECLARE
    v_name VARCHAR2(100) := 'Mahi Prajapati';
    v_roll NUMBER := 34;
    v_dob DATE := TO_DATE(
        '15-03-2008',
        'DD-MM-YYYY'
    );

    v_pnr VARCHAR2(20);
    v_jdate DATE;
    v_base_fare NUMBER;
    v_age NUMBER;
    v_final_fare NUMBER;
BEGIN
    v_pnr :=
        'GJ' ||
        TO_CHAR(SYSDATE, 'YYMM') ||
        LPAD(v_roll, 4, '0');

    v_jdate :=
        SYSDATE +
        (MOD(v_roll, 30) + 1);

    v_base_fare :=
        v_roll * 50 + 200;

    v_age := TRUNC(
        MONTHS_BETWEEN(
            SYSDATE,
            v_dob
        ) / 12
    );

    IF v_age >= 60 THEN
        v_final_fare :=
            v_base_fare * 0.60;
    ELSE
        v_final_fare := v_base_fare;
    END IF;

    DBMS_OUTPUT.PUT_LINE(
        '================================'
    );
    DBMS_OUTPUT.PUT_LINE(
        'IRCTC RAILWAY TICKET'
    );
    DBMS_OUTPUT.PUT_LINE(
        '================================'
    );
    DBMS_OUTPUT.PUT_LINE(
        'Passenger: ' || v_name
    );
    DBMS_OUTPUT.PUT_LINE(
        'PNR: ' || v_pnr
    );
    DBMS_OUTPUT.PUT_LINE(
        'Age: ' || v_age
    );
    DBMS_OUTPUT.PUT_LINE(
        'Journey: ' ||
        TO_CHAR(
            v_jdate,
            'Day, DD-MON-YYYY'
        )
    );
    DBMS_OUTPUT.PUT_LINE(
        'Base Fare: Rs.' ||
        v_base_fare
    );
    DBMS_OUTPUT.PUT_LINE(
        'Final Fare: Rs.' ||
        v_final_fare
    );
    DBMS_OUTPUT.PUT_LINE(
        '================================'
    );
END;
