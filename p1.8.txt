P1.8  Student ID Card Generator — String Functions   [Hard]  [SUBSTR, INSTR, UPPER, LPAD, RPAD,
LENGTH, TO_CHAR, REPLACE] Problem Statement:  Build a "LJICA Student ID Card Generator".
Input: full_name, roll_no, dob. Steps: (1) Convert name to UPPER, (2) Extract first name
using SUBSTR(name, 1, INSTR(name," ")-1), (3) Generate ID = "LJICA"||TO_CHAR(dob,"YYYY")||
LPAD(roll_no,3,"0"), (4) Calculate age from DOB, (5) Print formatted card using RPAD/LPAD
for alignment (like a printed ID card). Bonus: if LENGTH(full_name)>20 then abbreviate
last name to first initial.


-->answer

DECLARE
    v_name VARCHAR2(100) := 'Mahi Prajapati';
    v_roll NUMBER := 34;
    v_dob DATE := TO_DATE('15-03-2008', 'DD-MM-YYYY');

    v_upper VARCHAR2(100);
    v_first VARCHAR2(50);
    v_id VARCHAR2(50);
    v_age NUMBER;
BEGIN
    v_upper := UPPER(v_name);

    v_first := SUBSTR(
        v_name,
        1,
        INSTR(v_name, ' ') - 1
    );

    v_id := 'LJICA' ||
            TO_CHAR(v_dob, 'YYYY') ||
            LPAD(v_roll, 3, '0');

    v_age := TRUNC(
        MONTHS_BETWEEN(SYSDATE, v_dob) / 12
    );

    DBMS_OUTPUT.PUT_LINE(
        '========================================'
    );
    DBMS_OUTPUT.PUT_LINE(
        RPAD('LJICA STUDENT ID CARD', 40)
    );
    DBMS_OUTPUT.PUT_LINE(
        '========================================'
    );
    DBMS_OUTPUT.PUT_LINE('Name    : ' || v_upper);
    DBMS_OUTPUT.PUT_LINE('First   : ' || v_first);
    DBMS_OUTPUT.PUT_LINE('Roll No : ' || v_roll);
    DBMS_OUTPUT.PUT_LINE('ID      : ' || v_id);
    DBMS_OUTPUT.PUT_LINE('DOB     : ' ||
                         TO_CHAR(v_dob, 'DD-MON-YYYY'));
    DBMS_OUTPUT.PUT_LINE('Age     : ' || v_age);
    DBMS_OUTPUT.PUT_LINE(
        '========================================'
    );
END;
