P1.3  Constants, NULL and NVL   [Easy]  [CONSTANT, NULL, NVL, NVL2, GST calculation]
Problem Statement:  Part A: Declare CONSTANT c_gst_rate NUMBER := 18. Calculate GST 
breakdown: CGST=9%, SGST=9% on base_price. Print: base price, CGST amount, SGST 
amount, total. Part B: Declare v_city VARCHAR2(50) := NULL. Use NVL to replace with 
"Ahmedabad". Use NVL2: if city not null print "City Known: [city]" else print "City Unknown". Show 
difference between NVL and NVL2 clearly.


SET SERVEROUTPUT ON;

DECLARE
    c_gst_rate CONSTANT NUMBER := 18;
    v_roll NUMBER := 34;
    v_base_price NUMBER;
    v_cgst NUMBER;
    v_sgst NUMBER;
    v_total NUMBER;
    v_city VARCHAR2(50) := NULL;
BEGIN
    v_base_price := v_roll * 500 + 1000;
    v_cgst := v_base_price * 9 / 100;
    v_sgst := v_base_price * 9 / 100;
    v_total := v_base_price + v_cgst + v_sgst;

    DBMS_OUTPUT.PUT_LINE('Base Price: Rs.' || v_base_price);
    DBMS_OUTPUT.PUT_LINE('CGST: Rs.' || v_cgst);
    DBMS_OUTPUT.PUT_LINE('SGST: Rs.' || v_sgst);
    DBMS_OUTPUT.PUT_LINE('Total: Rs.' || v_total);

    DBMS_OUTPUT.PUT_LINE('NVL City: ' || NVL(v_city, 'Ahmedabad'));

    IF v_city IS NOT NULL THEN
        DBMS_OUTPUT.PUT_LINE('NVL2: City Known: ' || v_city);
    ELSE
        DBMS_OUTPUT.PUT_LINE('NVL2: City Unknown');
    END IF;
END;
