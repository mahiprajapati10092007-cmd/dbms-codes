P1.12  Result Card with BOOLEAN and NVL   [Easy]  [BOOLEAN, NVL, comparison, CASE expression]
Problem Statement:  Declare marks for 5 subjects: v_m1..v_m5 NUMBER, where v_m5 is NULL
(student was absent). Use NVL(v_m5, 0) while totalling. Calculate total and percentage
(out of 500). Declare v_passed BOOLEAN := (percentage >= 40). Print total, percentage and 
result. Remember: you cannot pass a BOOLEAN directly to DBMS_OUTPUT.PUT_LINE —
use a CASE expression or IF to convert it to text. Show this limitation in a comment.





DECLARE
    v_roll NUMBER := 34;
    v_mark NUMBER;

    v_m1 NUMBER;
    v_m2 NUMBER;
    v_m3 NUMBER;
    v_m4 NUMBER;
    v_m5 NUMBER := NULL;

    v_total NUMBER;
    v_pct NUMBER;
    v_passed BOOLEAN;
BEGIN
    v_mark := MOD(v_roll, 30) + 60;

    v_m1 := v_mark;
    v_m2 := v_mark;
    v_m3 := v_mark;
    v_m4 := v_mark;

    v_total :=
        v_m1 + v_m2 + v_m3 + v_m4 +
        NVL(v_m5, 0);

    v_pct := ROUND(
        v_total / 500 * 100,
        2
    );

    v_passed := v_pct >= 40;

    DBMS_OUTPUT.PUT_LINE('Total: ' || v_total);
    DBMS_OUTPUT.PUT_LINE(
        'Percentage: ' || v_pct || '%'
    );

    -- BOOLEAN cannot be printed directly.
    IF v_passed THEN
        DBMS_OUTPUT.PUT_LINE('Result: PASS');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Result: FAIL');
    END IF;
END;
