SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_book IS
        SELECT title, price
        FROM book
        WHERE price > 500;
BEGIN
    OPEN c_book;

    LOOP
        FETCH c_book INTO v_title, v_price;
        EXIT WHEN c_book%NOTFOUND;

        DBMS_OUTPUT.PUT_LINE(
            c_book%ROWCOUNT || '. ' ||
            v_title || ' - Rs. ' || v_price
        );
    END LOOP;

    CLOSE c_book;
END;
/
