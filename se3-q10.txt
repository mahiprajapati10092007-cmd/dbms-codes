SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_issue(p_days NUMBER) IS
        SELECT issue_id, book_id, issue_date
        FROM book_issue
        WHERE return_date IS NULL
        AND (TRUNC(SYSDATE) - issue_date) > p_days;

    v_days NUMBER := &days;
    v_overdue_days NUMBER;
    v_fine NUMBER(10,2);
    v_total NUMBER(12,2) := 0;
BEGIN
    FOR r IN c_issue(v_days) LOOP
        v_overdue_days :=
            (TRUNC(SYSDATE) - r.issue_date) - v_days;

        v_fine := v_overdue_days * 2;
        v_total := v_total + v_fine;

        DBMS_OUTPUT.PUT_LINE(
            'Issue ID: ' || r.issue_id ||
            ', Fine = Rs. ' || v_fine
        );
    END LOOP;

    DBMS_OUTPUT.PUT_LINE(
        'Total Fine Payable = Rs. ' || v_total
    );
END;
/
