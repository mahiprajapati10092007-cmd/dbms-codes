SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_book IS
        SELECT title, price
        FROM book
        ORDER BY price DESC;

    v_title book.title%TYPE;
    v_price book.price%TYPE;
BEGIN
    OPEN c_book;

    LOOP
        FETCH c_book INTO v_title, v_price;
        EXIT WHEN c_book%NOTFOUND;

        DBMS_OUTPUT.PUT_LINE(
            c_book%ROWCOUNT || '. ' ||
            v_title || ' - Rs. ' || v_price
        );

        EXIT WHEN c_book%ROWCOUNT = 5;
    END LOOP;

    CLOSE c_book;
END;
/
