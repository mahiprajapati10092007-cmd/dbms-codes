P1.21  Salary Slip Generator with %ROWTYPE   [Hard]  [%ROWTYPE, SELECT INTO, arithmetic, 
TO_CHAR, RPAD/LPAD] Problem Statement:  Generate a monthly salary slip. Fetch the full 
employee row into v_emp employees%ROWTYPE for the seeded employee. Compute: DA = 40% of 
salary, HRA = 20% of salary, PF deduction = 12% of salary, Net Pay = salary + DA + HRA - PF. 
Print a professional slip: header with company name and TO_CHAR(SYSDATE,'Month YYYY'), 
employee name/ID/job, then each earning and deduction right-aligned using LPAD with 
TO_CHAR(amount,'99,99,999.99'), and Net Pay at bottom. All amounts must line up in one 
column.
DECLARE
    v_roll NUMBER := 34;
    v_id NUMBER;

    v_emp employees%ROWTYPE;

    v_da NUMBER;
    v_hra NUMBER;
    v_pf NUMBER;
    v_net NUMBER;
BEGIN
    v_id := 100 + MOD(v_roll, 20);

    SELECT *
    INTO v_emp
    FROM employees
    WHERE employee_id = v_id;

    v_da := v_emp.salary * 0.40;
    v_hra := v_emp.salary * 0.20;
    v_pf := v_emp.salary * 0.12;

    v_net :=
        v_emp.salary +
        v_da +
        v_hra -
        v_pf;

    DBMS_OUTPUT.PUT_LINE(
        '========================================'
    );
    DBMS_OUTPUT.PUT_LINE(
        '        COMPANY SALARY SLIP'
    );
    DBMS_OUTPUT.PUT_LINE(
        TO_CHAR(SYSDATE, 'Month YYYY')
    );
    DBMS_OUTPUT.PUT_LINE(
        '========================================'
    );

    DBMS_OUTPUT.PUT_LINE(
        'Employee: ' ||
        v_emp.first_name || ' ' ||
        v_emp.last_name
    );
    DBMS_OUTPUT.PUT_LINE(
        'Employee ID: ' ||
        v_emp.employee_id
    );
    DBMS_OUTPUT.PUT_LINE(
        'Job: ' || v_emp.job_id
    );

    DBMS_OUTPUT.PUT_LINE(
        RPAD('Basic Salary', 20) ||
        LPAD(
            TO_CHAR(
                v_emp.salary,
                '99,99,999.99'
            ),
            15
        )
    );

    DBMS_OUTPUT.PUT_LINE(
        RPAD('DA', 20) ||
        LPAD(
            TO_CHAR(
                v_da,
                '99,99,999.99'
            ),
            15
        )
    );

    DBMS_OUTPUT.PUT_LINE(
        RPAD('HRA', 20) ||
        LPAD(
            TO_CHAR(
                v_hra,
                '99,99,999.99'
            ),
            15
        )
    );

    DBMS_OUTPUT.PUT_LINE(
        RPAD('PF Deduction', 20) ||
        LPAD(
            TO_CHAR(
                v_pf,
                '99,99,999.99'
            ),
            15
        )
    );

    DBMS_OUTPUT.PUT_LINE(
        '----------------------------------------'
    );

    DBMS_OUTPUT.PUT_LINE(
        RPAD('NET PAY', 20) ||
        LPAD(
            TO_CHAR(
                v_net,
                '99,99,999.99'
            ),
            15
        )
    );
END;
