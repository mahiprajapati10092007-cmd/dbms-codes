SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_member(p_letter VARCHAR2) IS
        SELECT member_name
        FROM lib_member
        WHERE UPPER(member_name)
              LIKE UPPER(p_letter) || '%';

    v_letter VARCHAR2(1) := '&letter';
BEGIN
    FOR r IN c_member(v_letter) LOOP
        DBMS_OUTPUT.PUT_LINE(r.member_name);
    END LOOP;
END;
/
