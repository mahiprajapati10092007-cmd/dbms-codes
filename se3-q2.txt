SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_book(p_pub VARCHAR2) IS
        SELECT b.title, b.price
        FROM book b
        JOIN publisher p
        ON b.pub_id = p.pub_id
        WHERE UPPER(p.pub_name) = UPPER(p_pub);

    v_publisher VARCHAR2(40) := '&publisher_name';
BEGIN
    FOR r IN c_book(v_publisher) LOOP
        DBMS_OUTPUT.PUT_LINE(
            r.title || ' - Rs. ' || r.price
        );
    END LOOP;
END;
