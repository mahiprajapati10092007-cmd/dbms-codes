P1.18  Messy Name Cleaner   [Medium]  [TRIM, INITCAP, REPLACE, LOWER, LENGTH, SUBSTR, INSTR]
 Problem Statement:  Admission forms arrive with messy names like "  pankti   PATEL " 
 (extra spaces, wrong case). Write a block that: (1) TRIMs leading/trailing spaces, 
 (2) collapses double spaces using REPLACE(name, '  ', ' ') — apply twice and explain why 
 once may not be enough, (3) applies INITCAP, (4) extracts first and last name using 
 SUBSTR/INSTR, (5) generates college email = LOWER(firstname || '.' || lastname || 
 '@ljku.edu.in'). Print cleaned name, first, last, email and LENGTH before vs after 
 cleaning


DECLARE
    v_raw VARCHAR2(100) :=
        '  MAHI   PRAJAPATI  ';

    v_clean VARCHAR2(100);
    v_first VARCHAR2(50);
    v_last VARCHAR2(50);
    v_email VARCHAR2(100);
BEGIN
    v_clean := TRIM(v_raw);

    -- Replace double spaces more than once
    -- because multiple spaces may remain.
    v_clean := REPLACE(v_clean, '  ', ' ');
    v_clean := REPLACE(v_clean, '  ', ' ');

    v_clean := INITCAP(v_clean);

    v_first := SUBSTR(
        v_clean,
        1,
        INSTR(v_clean, ' ') - 1
    );

    v_last := SUBSTR(
        v_clean,
        INSTR(v_clean, ' ') + 1
    );

    v_email := LOWER(
        v_first || '.' ||
        v_last || '@ljku.edu.in'
    );

    DBMS_OUTPUT.PUT_LINE(
        'Original: ' || v_raw
    );
    DBMS_OUTPUT.PUT_LINE(
        'Cleaned: ' || v_clean
    );
    DBMS_OUTPUT.PUT_LINE(
        'First Name: ' || v_first
    );
    DBMS_OUTPUT.PUT_LINE(
        'Last Name: ' || v_last
    );
    DBMS_OUTPUT.PUT_LINE(
        'Email: ' || v_email
    );
    DBMS_OUTPUT.PUT_LINE(
        'Length Before: ' || LENGTH(v_raw)
    );
    DBMS_OUTPUT.PUT_LINE(
        'Length After: ' || LENGTH(v_clean)
    );
END;
