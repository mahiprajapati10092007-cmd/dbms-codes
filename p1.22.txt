P1.22  Kirana Store Bill with NULL Coupon   [Hard]  [NVL, NVL2, CONSTANT, arithmetic, 
TO_CHAR, rounding] Problem Statement:  A kirana store bill: v_qty items at v_price each. 
A coupon variable v_coupon NUMBER may be NULL (no coupon) or a percentage. Compute: gross =
 qty*price, discount = gross * NVL(v_coupon,0)/100, taxable = gross - discount, CGST = SGST =
  9% of taxable, final total ROUNDed to nearest rupee. Use NVL2(v_coupon, 'Coupon applied: '||
  v_coupon||'%', 'No coupon') in the printout. Run the block twice — once with coupon NULL, 
  once with coupon = 10 — and show both bills.
DECLARE
    v_roll NUMBER := 34;

    v_qty NUMBER;
    v_price NUMBER;
    v_coupon NUMBER := NULL;

    v_gross NUMBER;
    v_disc NUMBER;
    v_taxable NUMBER;
    v_cgst NUMBER;
    v_sgst NUMBER;
    v_total NUMBER;
BEGIN
    v_qty := MOD(v_roll, 5) + 2;
    v_price := v_roll * 100 + 400;

    v_gross := v_qty * v_price;

    v_disc :=
        v_gross *
        NVL(v_coupon, 0) / 100;

    v_taxable :=
        v_gross - v_disc;

    v_cgst :=
        v_taxable * 9 / 100;

    v_sgst :=
        v_taxable * 9 / 100;

    v_total := ROUND(
        v_taxable +
        v_cgst +
        v_sgst
    );

    DBMS_OUTPUT.PUT_LINE(
        '----- KIRANA STORE BILL -----'
    );
    DBMS_OUTPUT.PUT_LINE(
        'Quantity: ' || v_qty
    );
    DBMS_OUTPUT.PUT_LINE(
        'Price: Rs.' || v_price
    );
    DBMS_OUTPUT.PUT_LINE(
        'Gross: Rs.' || v_gross
    );
    DBMS_OUTPUT.PUT_LINE(
        NVL2(
            v_coupon,
            'Coupon applied: ' ||
            v_coupon || '%',
            'No coupon'
        )
    );
    DBMS_OUTPUT.PUT_LINE(
        'Discount: Rs.' || v_disc
    );
    DBMS_OUTPUT.PUT_LINE(
        'CGST: Rs.' || v_cgst
    );
    DBMS_OUTPUT.PUT_LINE(
        'SGST: Rs.' || v_sgst
    );
    DBMS_OUTPUT.PUT_LINE(
        'Final Total: Rs.' || v_total
    );
END;
