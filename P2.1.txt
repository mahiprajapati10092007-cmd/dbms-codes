P2.1  Complete Grade Card System   [Easy]  [IF-THEN-ELSIF, nested IF, Boolean conditions] Problem Statement:  Input marks for 5 subjects (use variables). Calculate total and percentage. Use IFELSIF for grade: >=90=A+, >=80=A, >=70=B, >=60=C, >=50=D, <50=FAIL. Rules using nested IF: (1) If any subject < 35 marks: DETAINED — print which subject failed. (2) If all pass: print class rank: Distinction(>=75%), First(>=60%), Second(>=50%), Pass(<50% but all sub>=35). Print full report card with all 5 marks, total, percentage, grade, result. 




SET SERVEROUTPUT ON;

DECLARE
    v_roll_no NUMBER := 34;
    v_name VARCHAR2(50) := 'Mahi Prajapati';

    s1 NUMBER := 85;
    s2 NUMBER := 78;
    s3 NUMBER := 92;
    s4 NUMBER := 65;
    s5 NUMBER := 88;

    v_total NUMBER;
    v_percentage NUMBER;
    v_grade VARCHAR2(10);
    v_result VARCHAR2(20);
    v_class VARCHAR2(30);
BEGIN
    v_total := s1 + s2 + s3 + s4 + s5;
    v_percentage := v_total / 5;

    IF s1 < 35 OR s2 < 35 OR s3 < 35 OR s4 < 35 OR s5 < 35 THEN
        v_result := 'DETAINED';

        IF s1 < 35 THEN
            DBMS_OUTPUT.PUT_LINE('Failed Subject 1');
        END IF;
        IF s2 < 35 THEN
            DBMS_OUTPUT.PUT_LINE('Failed Subject 2');
        END IF;
        IF s3 < 35 THEN
            DBMS_OUTPUT.PUT_LINE('Failed Subject 3');
        END IF;
        IF s4 < 35 THEN
            DBMS_OUTPUT.PUT_LINE('Failed Subject 4');
        END IF;
        IF s5 < 35 THEN
            DBMS_OUTPUT.PUT_LINE('Failed Subject 5');
        END IF;
    ELSE
        v_result := 'PASS';

        IF v_percentage >= 75 THEN
            v_class := 'Distinction';
        ELSIF v_percentage >= 60 THEN
            v_class := 'First Class';
        ELSIF v_percentage >= 50 THEN
            v_class := 'Second Class';
        ELSE
            v_class := 'Pass Class';
        END IF;
    END IF;

    IF v_percentage >= 90 THEN
        v_grade := 'A+';
    ELSIF v_percentage >= 80 THEN
        v_grade := 'A';
    ELSIF v_percentage >= 70 THEN
        v_grade := 'B';
    ELSIF v_percentage >= 60 THEN
        v_grade := 'C';
    ELSIF v_percentage >= 50 THEN
        v_grade := 'D';
    ELSE
        v_grade := 'FAIL';
    END IF;

    DBMS_OUTPUT.PUT_LINE('Name: ' || v_name);
    DBMS_OUTPUT.PUT_LINE('Roll No: ' || v_roll_no);
    DBMS_OUTPUT.PUT_LINE('Marks: ' || s1 || ', ' || s2 || ', ' || s3 || ', ' || s4 || ', ' || s5);
    DBMS_OUTPUT.PUT_LINE('Total: ' || v_total);
    DBMS_OUTPUT.PUT_LINE('Percentage: ' || v_percentage || '%');
    DBMS_OUTPUT.PUT_LINE('Grade: ' || v_grade);
    DBMS_OUTPUT.PUT_LINE('Result: ' || v_result);
    DBMS_OUTPUT.PUT_LINE('Class: ' || v_class);
END;
