SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_book IS
        SELECT title, stock
        FROM book
        WHERE category = 'Database'
        FOR UPDATE;

    v_title book.title%TYPE;
    v_stock book.stock%TYPE;
BEGIN
    OPEN c_book;

    LOOP
        FETCH c_book INTO v_title, v_stock;
        EXIT WHEN c_book%NOTFOUND;

        UPDATE book
        SET stock = stock + 10
        WHERE CURRENT OF c_book;

        DBMS_OUTPUT.PUT_LINE(
            v_title || ': Old Stock = ' ||
            v_stock || ', New Stock = ' ||
            (v_stock + 10)
        );
    END LOOP;

    CLOSE c_book;

    COMMIT;
END;
