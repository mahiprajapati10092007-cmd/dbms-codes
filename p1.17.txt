P1.17  Home Loan EMI Calculator   [Medium]  [POWER, ROUND, arithmetic, TO_CHAR] Problem 
Statement:  Build an SBI home loan EMI calculator. Variables: v_p (principal), v_annual_rate 
:= 9, v_years := 5. Compute monthly rate r = annual/12/100 and n = years*12. EMI = P * r * POWER(1+r, n)
 / (POWER(1+r, n) - 1). Also compute total payment = EMI * n and total interest = total - 
 P. Print all three with TO_CHAR comma formatting. Bonus: print what % of your total payment
  is pure interest.


DECLARE
    v_roll NUMBER := 34;
    v_p NUMBER;

    v_annual_rate NUMBER := 9;
    v_years NUMBER := 5;

    v_r NUMBER;
    v_n NUMBER;
    v_emi NUMBER;
    v_total NUMBER;
    v_interest NUMBER;
    v_interest_pct NUMBER;
BEGIN
    v_p := v_roll * 10000 + 50000;

    v_r := v_annual_rate / 12 / 100;
    v_n := v_years * 12;

    v_emi := ROUND(
        v_p * v_r *
        POWER(1 + v_r, v_n) /
        (POWER(1 + v_r, v_n) - 1),
        2
    );

    v_total := v_emi * v_n;
    v_interest := v_total - v_p;

    v_interest_pct := ROUND(
        v_interest / v_total * 100,
        2
    );

    DBMS_OUTPUT.PUT_LINE(
        'Principal: Rs.' ||
        TO_CHAR(v_p, '99,99,999')
    );
    DBMS_OUTPUT.PUT_LINE(
        'Monthly EMI: Rs.' ||
        TO_CHAR(v_emi, '99,99,999.99')
    );
    DBMS_OUTPUT.PUT_LINE(
        'Total Payment: Rs.' ||
        TO_CHAR(v_total, '99,99,999.99')
    );
    DBMS_OUTPUT.PUT_LINE(
        'Total Interest: Rs.' ||
        TO_CHAR(v_interest, '99,99,999.99')
    );
    DBMS_OUTPUT.PUT_LINE(
        'Interest Percentage: ' ||
        v_interest_pct || '%'
    );
END;
