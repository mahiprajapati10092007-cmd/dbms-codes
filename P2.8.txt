P2.8  Loan EMI Affordability Checker   [Hard]  [LOOP + IF + CASE + financial formula, real banking logic] Problem Statement:  HDFC/SBI style loan portal: Input: monthly_salary, existing_emi, loan_amount, annual_rate, tenure_months. Rules: (1) FOIR limit = 40% of salary (max total EMI), (2) Min salary = Rs.25,000 for any loan, (3) Max loan = 60x monthly salary. EMI formula = P*r*(1+r)^n/((1+r)^n-1) where r=monthly_rate. If EMI > FOIR: use WHILE LOOP to find max eligible loan (reduce by Rs.10,000 each iteration). Print: calculated EMI, FOIR limit, approved amount, decision using CASE ("APPROVED"/"CONDITIONAL"/"REJECTED"), monthly cost breakdown. 


//answer

SET SERVEROUTPUT ON;

DECLARE
    v_monthly_salary NUMBER := 50000;
    v_existing_emi NUMBER := 5000;
    v_loan_amount NUMBER := 1000000;
    v_annual_rate NUMBER := 10;
    v_tenure NUMBER := 60;

    v_monthly_rate NUMBER;
    v_emi NUMBER;
    v_foir_limit NUMBER;
    v_max_loan NUMBER;
    v_approved_amount NUMBER;
    v_decision VARCHAR2(20);
BEGIN
    v_monthly_rate := v_annual_rate / 12 / 100;
    v_foir_limit := v_monthly_salary * 0.40;
    v_max_loan := v_monthly_salary * 60;

    -- Check maximum loan limit
    IF v_loan_amount > v_max_loan THEN
        v_loan_amount := v_max_loan;
    END IF;

    -- EMI calculation
    v_emi := v_loan_amount * v_monthly_rate *
             POWER(1 + v_monthly_rate, v_tenure) /
             (POWER(1 + v_monthly_rate, v_tenure) - 1);

    v_approved_amount := v_loan_amount;

    -- Find affordable loan
    WHILE v_existing_emi + v_emi > v_foir_limit
          AND v_approved_amount > 0 LOOP

        v_approved_amount := v_approved_amount - 10000;

        v_emi := v_approved_amount * v_monthly_rate *
                 POWER(1 + v_monthly_rate, v_tenure) /
                 (POWER(1 + v_monthly_rate, v_tenure) - 1);
    END LOOP;

    -- Decision
    CASE
        WHEN v_monthly_salary < 25000 THEN
            v_decision := 'REJECTED';

        WHEN v_approved_amount = 0 THEN
            v_decision := 'REJECTED';

        WHEN v_approved_amount < v_loan_amount THEN
            v_decision := 'CONDITIONAL';

        ELSE
            v_decision := 'APPROVED';
    END CASE;

    DBMS_OUTPUT.PUT_LINE('Monthly Salary: Rs.' || v_monthly_salary);
    DBMS_OUTPUT.PUT_LINE('Existing EMI: Rs.' || v_existing_emi);
    DBMS_OUTPUT.PUT_LINE('FOIR Limit: Rs.' || v_foir_limit);
    DBMS_OUTPUT.PUT_LINE('Requested Loan: Rs.' || v_loan_amount);
    DBMS_OUTPUT.PUT_LINE('Approved Loan: Rs.' || v_approved_amount);
    DBMS_OUTPUT.PUT_LINE('Calculated EMI: Rs.' || ROUND(v_emi));
    DBMS_OUTPUT.PUT_LINE('Total Monthly EMI: Rs.' || ROUND(v_existing_emi + v_emi));
    DBMS_OUTPUT.PUT_LINE('Decision: ' || v_decision);
END;
